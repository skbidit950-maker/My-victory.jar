# DonutClient — Forge 1.21.11 / 61.2.5

Independent client-side utility mod for Java 21. This is not the original LarpClient source.

## Implemented catalog entries in v0.6.0 (19/176)

- Fullbright (`G`) with restoration of the previous gamma.
- Zoom (`Z`) with restoration of the previous FOV.
- Auto Sprint while forward is held, when not crouching/passenger and with sufficient hunger.
- AutoEat: uses food already held in the main hand or offhand when hunger is low. It does not search or rearrange inventory.
- AutoTool: selects the fastest suitable tool already in the hotbar for the block being targeted.
- Custom center crosshair.
- Custom FOV toggle (100 degrees; restores the previous setting when disabled).
- Performance HUD: client FPS, server ping when connected, and average worn-armor durability.
- Target HUD: name and health of the entity under the crosshair.
- Looked-at entity label.
- HUD coordinates and dimension.
- Target block name/coordinates and block/sky light display.
- Local waypoints (`K` saves; `L` clears).
- Periodic scan of already-loaded nearby chunks for heuristic SusChunkFinder/Base Finder clues, storage blocks and spawners.
- Motion Scanner heuristic label.
- Config persistence for implemented module toggles.

## Keybinds

| Key | Action |
|---|---|
| Right Shift | Open ClickGUI |
| G | Toggle Fullbright |
| Z | Toggle Zoom |
| K | Save waypoint |
| L | Clear waypoints |

Other implemented modules are toggled in ClickGUI.

## Limitations and safety

- **19 of 176 catalog entries are marked implemented.** Some are basic informational implementations, not complete equivalents of similarly named commercial modules.
- Chunk scanning examines only chunks already loaded by the client, in a small radius and vertical band. It does not load chunks or reveal unloaded terrain.
- Base/SusChunk/Motion findings are heuristics, not proof of a base or player activity.
- StorageESP/SpawnerFinder currently provide scan counts/findings, not world-space 3D boxes.
- Freecam, Freelook, and 3D ESP boxes are not implemented.
- Most Combat, Movement, Player automation, Cosmetics and DonutSMP modules remain planned.
- No anti-cheat bypasses, packet spoofing, combat automation, or server-side exploits are included.
- **Not compiled or tested against Forge 61.2.5 in this environment.** Gradle is not installed in this runtime, so API compatibility and in-game behavior still require a local `gradle build` and client launch using JDK 21.

package dev.donutclient;

import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.client.gui.screens.Screen;
import net.minecraft.network.chat.Component;

import java.util.List;

public final class ClickGuiScreen extends Screen {
    private static final int PANEL_X = 24;
    private static final int PANEL_Y = 24;
    private static final int PANEL_W = 360;
    private static final int ROW_H = 22;
    private int scroll;

    public ClickGuiScreen() {
        super(Component.literal("DonutClient"));
    }

    @Override
    public void render(GuiGraphics g, int mouseX, int mouseY, float partialTick) {
        renderBackground(g, mouseX, mouseY, partialTick);
        int x2 = PANEL_X + PANEL_W;
        int y2 = Math.min(height - 20, PANEL_Y + 48 + ModuleRegistry.all().size() * ROW_H);
        g.fill(PANEL_X, PANEL_Y, x2, Math.max(PANEL_Y + 100, y2), 0xE9151820);
        g.fill(PANEL_X, PANEL_Y, x2, PANEL_Y + 28, 0xFF273449);
        g.drawString(font, "DONUTCLIENT  |  " + ModuleRegistry.all().size() + " MODULES",
                PANEL_X + 10, PANEL_Y + 10, 0xFFFFFFFF, false);
        g.drawString(font, "Green = implemented   Gray = catalog/backlog",
                PANEL_X + 10, PANEL_Y + 34, 0xFFB8C4D8, false);

        int start = Math.max(0, scroll);
        int visible = Math.max(1, (height - PANEL_Y - 58) / ROW_H);
        List<ClientModule> modules = ModuleRegistry.all();
        for (int i = start; i < Math.min(modules.size(), start + visible); i++) {
            ClientModule m = modules.get(i);
            int y = PANEL_Y + 54 + (i - start) * ROW_H;
            int bg = m.implemented() ? (m.enabled() ? 0xFF244D3C : 0xFF202A35) : 0xFF303039;
            g.fill(PANEL_X + 6, y, x2 - 6, y + ROW_H - 2, bg);
            String status = m.implemented() ? (m.enabled() ? "ON" : "OFF") : "PLANNED";
            g.drawString(font, m.category() + "  /  " + m.name(),
                    PANEL_X + 12, y + 6, m.implemented() ? 0xFFE8EDF5 : 0xFF969AA3, false);
            int sw = font.width(status);
            g.drawString(font, status, x2 - 14 - sw, y + 6,
                    m.implemented() ? 0xFFB8F2C9 : 0xFFB0B0B0, false);
        }
        super.render(g, mouseX, mouseY, partialTick);
    }

    @Override
    public boolean mouseClicked(double mouseX, double mouseY, int button) {
        if (button == 0 && mouseX >= PANEL_X && mouseX <= PANEL_X + PANEL_W
                && mouseY >= PANEL_Y + 54) {
            int row = (int) ((mouseY - (PANEL_Y + 54)) / ROW_H);
            int index = scroll + row;
            if (index >= 0 && index < ModuleRegistry.all().size()) {
                ModuleRegistry.all().get(index).toggle();
                return true;
            }
        }
        return super.mouseClicked(mouseX, mouseY, button);
    }

    @Override
    public boolean mouseScrolled(double mouseX, double mouseY, double delta) {
        int max = Math.max(0, ModuleRegistry.all().size() -
                Math.max(1, (height - PANEL_Y - 58) / ROW_H));
        scroll = Math.max(0, Math.min(max, scroll - (int) Math.signum(delta) * 3));
        return true;
    }

    @Override
    public boolean isPauseScreen() {
        return false;
    }
}

package dev.donutclient;

import net.minecraftforge.fml.loading.FMLPaths;
import java.io.*;
import java.nio.file.*;
import java.util.Properties;

public final class ClientConfig {
    private static final Path FILE = FMLPaths.CONFIGDIR.get().resolve("donutclient.properties");
    private ClientConfig() {}

    public static void load() {
        if (!Files.exists(FILE)) return;
        Properties p = new Properties();
        try (Reader r = Files.newBufferedReader(FILE)) {
            p.load(r);
            for (ClientModule module : ModuleRegistry.all()) {
                if (module.implemented()) {
                    module.setEnabled(Boolean.parseBoolean(p.getProperty("module." + module.id(), "false")));
                }
            }
        } catch (IOException e) {
            System.err.println("[DonutClient] Config load failed: " + e.getMessage());
        }
    }

    public static void save() {
        Properties p = new Properties();
        for (ClientModule module : ModuleRegistry.all()) {
            if (module.implemented()) p.setProperty("module." + module.id(), Boolean.toString(module.enabled()));
        }
        try {
            Files.createDirectories(FILE.getParent());
            try (Writer w = Files.newBufferedWriter(FILE)) {
                p.store(w, "DonutClient settings");
            }
        } catch (IOException e) {
            System.err.println("[DonutClient] Config save failed: " + e.getMessage());
        }
    }
}

    @Mod.EventBusSubscriber(modid = DonutClient.MOD_ID, value = Dist.CLIENT)
    public static final class ForgeEvents {
        @SubscribeEvent
        public static void clientTick(TickEvent.ClientTickEvent event) {
            if (event.phase != TickEvent.Phase.END) return;
            Minecraft mc = Minecraft.getInstance();
            if (mc.player == null || mc.level == null) return;
            clientTicks++;

            while (clickGuiKey.consumeClick()) mc.setScreen(new ClickGuiScreen());
            while (fullbrightKey.consumeClick()) {
                ModuleRegistry.byId("fullbright").ifPresent(ClientModule::toggle);
            }
            while (addWaypointKey.consumeClick()) addWaypoint();
            while (clearWaypointsKey.consumeClick()) {
                waypoints.clear();
                mc.player.displayClientMessage(Component.literal("DonutClient: waypoints cleared"), true);
            }
            while (zoomKey.consumeClick()) {
                ModuleRegistry.byId("zoom").ifPresent(ClientModule::toggle);
            }
            if (enabled("auto_sprint") && mc.screen == null && mc.options.keyUp.isDown()
                    && !mc.player.isCrouching() && !mc.player.isPassenger() && mc.player.getFoodData().getFoodLevel() > 6) {
                mc.player.setSprinting(true);
            }

            // AutoEat only uses food already held in either hand and invokes vanilla use-item.
            if (enabled("autoeat") && mc.screen == null && mc.gameMode != null
                    && mc.player.getFoodData().getFoodLevel() <= 16
                    && mc.player.canEat(false) && clientTicks - lastEatTick >= 20) {
                ItemStack main = mc.player.getMainHandItem();
                ItemStack off = mc.player.getOffhandItem();
                if (main.getFoodProperties(mc.player) != null) {
                    mc.gameMode.useItem(mc.player, InteractionHand.MAIN_HAND);
                    lastEatTick = clientTicks;
                } else if (off.getFoodProperties(mc.player) != null) {
                    mc.gameMode.useItem(mc.player, InteractionHand.OFF_HAND);
                    lastEatTick = clientTicks;
                }
            }

            // AutoTool chooses the fastest tool already in the hotbar for the targeted block.
            if (enabled("autotool") && mc.screen == null
                    && mc.hitResult instanceof net.minecraft.world.phys.BlockHitResult blockHit
                    && mc.gameMode != null) {
                BlockState state = mc.level.getBlockState(blockHit.getBlockPos());
                int bestSlot = mc.player.getInventory().getSelectedSlot();
                float bestSpeed = mc.player.getMainHandItem().getDestroySpeed(state);
                for (int slot = 0; slot < 9; slot++) {
                    ItemStack candidate = mc.player.getInventory().getItem(slot);
                    float speed = candidate.getDestroySpeed(state);
                    if (speed > bestSpeed) {
                        bestSpeed = speed;
                        bestSlot = slot;
                    }
                }
                mc.player.getInventory().setSelectedSlot(bestSlot);
            }

            if (clientTicks - lastScanTick >= 100) {
                lastScanTick = clientTicks;
                scanLoadedChunks(mc);
            }
            updateTargetedBlock(mc);
        }

        @SubscribeEvent
        public static void renderOverlay(RenderGuiOverlayEvent.Post event) {
            Minecraft mc = Minecraft.getInstance();
            if (mc.player == null || mc.level == null) return;
            var gui = event.getGuiGraphics();
            if (enabled("customcrosshair")) {
                int cx = mc.getWindow().getGuiScaledWidth() / 2;
                int cy = mc.getWindow().getGuiScaledHeight() / 2;
                int color = 0xEFFFFFFF;
                gui.fill(cx - 5, cy, cx - 2, cy + 1, color);
                gui.fill(cx + 2, cy, cx + 5, cy + 1, color);
                gui.fill(cx, cy - 5, cx + 1, cy - 2, color);
                gui.fill(cx, cy + 2, cx + 1, cy + 5, color);
            }
            int x = 8;
            int y = 8;

            // Lightweight local status overlays; these read client state only.
            if (enabled("performance")) {
                int fps = Minecraft.getInstance().getFps();
                String pingText = "Ping: n/a";
                if (mc.getConnection() != null && mc.player != null) {
                    var info = mc.getConnection().getPlayerInfo(mc.player.getUUID());
                    if (info != null) pingText = "Ping: " + info.getLatency() + " ms";
                }
                gui.drawString(mc.font, "FPS: " + fps + "  " + pingText, x, y,
                        0xFFB7E8FF, true);
                y += 12;
            }
            if (enabled("targethud") && mc.hitResult instanceof EntityHitResult entityHit
                    && entityHit.getEntity() instanceof LivingEntity living) {
                String targetName = living.getDisplayName().getString();
                String targetHealth = String.format(java.util.Locale.ROOT, "%.1f / %.1f HP",
                        living.getHealth(), living.getMaxHealth());
                gui.drawString(mc.font, "Target: " + targetName + "  " + targetHealth,
                        x, y, 0xFFFFA6A6, true);
                y += 12;
            }
            if (enabled("nametags") && mc.hitResult instanceof EntityHitResult entityHit) {
                gui.drawString(mc.font, "Look: " + entityHit.getEntity().getName().getString(),
                        x, y, 0xFFE7D5FF, true);
                y += 12;
            }
            if (enabled("performance") && mc.player != null) {
                int armor = 0;
                int worn = 0;
                for (var stack : mc.player.getArmorSlots()) {
                    if (!stack.isEmpty()) {
                        worn++;
                        int max = stack.getMaxDamage();
                        if (max > 0) armor += Math.max(0, 100 - (stack.getDamageValue() * 100 / max));
                    }
                }
                if (worn > 0) {
                    gui.drawString(mc.font, "Armor avg: " + (armor / worn) + "%",
                            x, y, 0xFFD5F0C8, true);
                    y += 12;
                }
            }
            if (enabled("hud")) {
                gui.drawString(mc.font, "DonutClient", x, y, 0xFFFFFFFF, true);
                y += 12;
                gui.drawString(mc.font, "XYZ " + mc.player.blockPosition().getX() + " "
                        + mc.player.blockPosition().getY() + " " + mc.player.blockPosition().getZ(),
                        x, y, 0xFFE4EAF2, true);
                y += 12;
                gui.drawString(mc.font, "Dimension: " + mc.level.dimension().location(),
                        x, y, 0xFFE4EAF2, true);
                y += 12;
            }
            if (enabled("block_overlay") && targetedBlock != null) {
                BlockState state = mc.level.getBlockState(targetedBlock);
                String blockName = BuiltInRegistries.BLOCK.getKey(state.getBlock()).toString();
                gui.drawString(mc.font, blockName + " @ " + targetedBlock.getX() + ","
                        + targetedBlock.getY() + "," + targetedBlock.getZ(),
                        x, y, 0xFFFFD27A, true);
                y += 12;
            }
            if (enabled("light_debug") && targetedBlock != null) {
                int blockLight = mc.level.getBrightness(net.minecraft.world.level.LightLayer.BLOCK, targetedBlock);
                int skyLight = mc.level.getBrightness(net.minecraft.world.level.LightLayer.SKY, targetedBlock);
                gui.drawString(mc.font, "Light: block " + blockLight + " / sky " + skyLight,
                        x, y, 0xFFB7E8FF, true);
                y += 12;
            }
            if (enabled("waypoints")) {
                for (Waypoint waypoint : waypoints) {
                    if (!waypoint.dimension.equals(mc.level.dimension())) continue;
                    double distance = Math.sqrt(mc.player.distanceToSqr(
                            waypoint.x + 0.5, waypoint.y + 0.5, waypoint.z + 0.5));
                    gui.drawString(mc.font, waypoint.name + " [" + (int) distance + "m] "
                            + waypoint.x + " " + waypoint.y + " " + waypoint.z,
                            x, y, 0xFF8FE6C0, true);
                    y += 11;
                }
            }
            if (enabled("suschunkfinder") || enabled("base_finder")
                    || enabled("motion_scanner") || enabled("storageesp")
                    || enabled("spawnerfinder")) {
                int shown = 0;
                for (ChunkFinding finding : findings.values()) {
                    if (shown++ >= 8) break;
                    if (finding.dimension != mc.level.dimension()) continue;
                    String label = "Chunk " + finding.x + "," + finding.z + " score " + finding.score;
                    if (enabled("suschunkfinder") && finding.susScore > 0) {
                        gui.drawString(mc.font, "Sus: " + label, x, y, 0xFFFFC66D, true);
                        y += 11;
                    } else if (enabled("base_finder") && finding.baseScore > 0) {
                        gui.drawString(mc.font, "Base clue: " + label, x, y, 0xFFB5A0FF, true);
                        y += 11;
                    } else if (enabled("motion_scanner") && finding.lightScore > 0) {
                        gui.drawString(mc.font, "Light anomaly: " + label, x, y, 0xFF8EDFFF, true);
                        y += 11;
                    }
                }
            }
        }
    }

    private static boolean enabled(String id) {
        return ModuleRegistry.byId(id).map(ClientModule::enabled).orElse(false);
    }

    public static void onModuleChanged(ClientModule module) {
        if (module.id().equals("fullbright")) applyFullbright(module.enabled());
        if (module.id().equals("zoom")) applyZoom(module.enabled());
        if (module.id().equals("custom_fov")) applyCustomFov(module.enabled());
    }

    private static void applyZoom(boolean enabled) {
        Minecraft mc = Minecraft.getInstance();
        if (mc.options == null) return;
        if (enabled) {
            if (!capturedFov) {
                previousFov = mc.options.fov().get();
                capturedFov = true;
            }
            mc.options.fov().set(30);
        } else if (capturedFov) {
            mc.options.fov().set(previousFov);
            capturedFov = false;
        }
    }

    private static void applyCustomFov(boolean enabled) {
        Minecraft mc = Minecraft.getInstance();
        if (mc.options == null) return;
        if (enabled) {
            if (!capturedCustomFov) {
                previousCustomFov = mc.options.fov().get();
                capturedCustomFov = true;
            }
            mc.options.fov().set(100);
        } else if (capturedCustomFov) {
            mc.options.fov().set(previousCustomFov);
            capturedCustomFov = false;
        }
    }

    public static void applyFullbright(boolean enabled) {
        Minecraft mc = Minecraft.getInstance();
        if (mc.options == null) return;
        Options options = mc.options;
        if (enabled) {
            if (!capturedGamma) {
                previousGamma = options.gamma().get();
                capturedGamma = true;
            }
            options.gamma().set(16.0);
        } else if (capturedGamma) {
            options.gamma().set(previousGamma);
            capturedGamma = false;
        }
    }

    private static void addWaypoint() {
        Minecraft mc = Minecraft.getInstance();
        if (mc.player == null || mc.level == null) return;
        BlockPos pos = mc.player.blockPosition();
        waypoints.add(new Waypoint("Waypoint " + (waypoints.size() + 1),
                pos.getX(), pos.getY(), pos.getZ(), mc.level.dimension()));
        mc.player.displayClientMessage(Component.literal("DonutClient: waypoint saved at "
                + pos.getX() + " " + pos.getY() + " " + pos.getZ()), true);
    }

    private static void updateTargetedBlock(Minecraft mc) {
        if (mc.hitResult instanceof net.minecraft.world.phys.BlockHitResult hit
                && hit.getType() == net.minecraft.world.phys.HitResult.Type.BLOCK) {
            targetedBlock = hit.getBlockPos();
        } else {
            targetedBlock = null;
        }
    }

    private static void scanLoadedChunks(Minecraft mc) {
        if (mc.level == null || mc.player == null) return;
        ChunkPos center = mc.player.chunkPosition();
        ResourceKey<Level> dimension = mc.level.dimension();
        int radius = 2;
        int minY = Math.max(mc.level.getMinBuildHeight(), mc.player.blockPosition().getY() - 32);
        int maxY = Math.min(mc.level.getMaxBuildHeight(), mc.player.blockPosition().getY() + 48);

        for (int cx = center.x - radius; cx <= center.x + radius; cx++) {
            for (int cz = center.z - radius; cz <= center.z + radius; cz++) {
                LevelChunk chunk = mc.level.getChunkSource().getChunk(cx, cz, net.minecraft.world.level.chunk.ChunkStatus.FULL, false);
                if (chunk == null) continue;
                int sus = 0;
                int base = 0;
                int light = 0;
                int storage = 0;
                int spawners = 0;

                for (int x = 0; x < 16; x++) {
                    for (int z = 0; z < 16; z++) {
                        for (int y = minY; y < maxY; y++) {
                            BlockPos pos = new BlockPos((cx << 4) + x, y, (cz << 4) + z);
                            Block block = chunk.getBlockState(pos).getBlock();

                            if (isGrowthClue(block)) sus++;
                            if (isBaseClue(block)) base++;
                            if (isStorage(block)) storage++;
                            if (block == Blocks.SPAWNER) spawners++;
                        }
                    }
                }
                if (sus > 0 || base > 0 || storage > 0 || spawners > 0) {
                    int lightScore = Math.max(0, storage - 1) + Math.max(0, sus / 3);
                    findings.put(chunkKey(cx, cz), new ChunkFinding(cx, cz, dimension,
                            sus, base, lightScore, storage, spawners));
                } else {
                    findings.remove(chunkKey(cx, cz));
                }
            }
        }
        while (findings.size() > 256) {
            Iterator<Long> it = findings.keySet().iterator();
            it.next();
            it.remove();
        }
    }

    private static boolean isGrowthClue(Block b) {
        return b == Blocks.AMETHYST_CLUSTER || b == Blocks.LARGE_AMETHYST_BUD
                || b == Blocks.MEDIUM_AMETHYST_BUD || b == Blocks.SMALL_AMETHYST_BUD
                || b == Blocks.BAMBOO || b == Blocks.KELP || b == Blocks.KELP_PLANT
                || b == Blocks.SWEET_BERRY_BUSH || b == Blocks.POINTED_DRIPSTONE
                || b == Blocks.CAVE_VINES || b == Blocks.CAVE_VINES_PLANT;
    }

    private static boolean isBaseClue(Block b) {
        return isStorage(b) || b == Blocks.CRAFTING_TABLE || b == Blocks.FURNACE
                || b == Blocks.BLAST_FURNACE || b == Blocks.SMOKER
                || b == Blocks.ANVIL || b == Blocks.ENCHANTING_TABLE
                || b == Blocks.BREWING_STAND || b == Blocks.BEACON
                || b == Blocks.RESPAWN_ANCHOR || b == Blocks.ENDER_CHEST;
    }

    private static boolean isStorage(Block b) {
        return b == Blocks.CHEST || b == Blocks.TRAPPED_CHEST || b == Blocks.BARREL
                || b == Blocks.SHULKER_BOX || b == Blocks.WHITE_SHULKER_BOX
                || b == Blocks.ORANGE_SHULKER_BOX || b == Blocks.MAGENTA_SHULKER_BOX
                || b == Blocks.LIGHT_BLUE_SHULKER_BOX || b == Blocks.YELLOW_SHULKER_BOX
                || b == Blocks.LIME_SHULKER_BOX || b == Blocks.PINK_SHULKER_BOX
                || b == Blocks.GRAY_SHULKER_BOX || b == Blocks.LIGHT_GRAY_SHULKER_BOX
                || b == Blocks.CYAN_SHULKER_BOX || b == Blocks.PURPLE_SHULKER_BOX
                || b == Blocks.BLUE_SHULKER_BOX || b == Blocks.BROWN_SHULKER_BOX
                || b == Blocks.GREEN_SHULKER_BOX || b == Blocks.RED_SHULKER_BOX
                || b == Blocks.BLACK_SHULKER_BOX;
    }

    private static long chunkKey(int x, int z) {
        return ((long) x << 32) ^ (z & 0xffffffffL);
    }

    private record ChunkFinding(int x, int z, ResourceKey<Level> dimension,
                                int susScore, int baseScore, int lightScore,
                                int storageCount, int spawnerCount) {
        int score() { return susScore + baseScore + storageCount * 2 + spawnerCount * 5; }
    }

package dev.donutclient;

public final class ClientModule {
    private final String id;
    private final String name;
    private final String category;
    private final String description;
    private final boolean implemented;
    private boolean enabled;

    public ClientModule(String id, String name, String category, String description, boolean implemented) {
        this.id = id;
        this.name = name;
        this.category = category;
        this.description = description;
        this.implemented = implemented;
    }

    public String id() { return id; }
    public String name() { return name; }
    public String category() { return category; }
    public String description() { return description; }
    public boolean implemented() { return implemented; }
    public boolean enabled() { return enabled; }

    public void setEnabled(boolean value) {
        if (!implemented) return;
        enabled = value;
        ModuleRegistry.onStateChanged(this);
    }

    public void toggle() {
        if (implemented) setEnabled(!enabled);
    }
}

package dev.donutclient;

import com.google.gson.Gson;
import com.google.gson.reflect.TypeToken;
import net.minecraftforge.fml.loading.FMLPaths;

import java.io.Reader;
import java.lang.reflect.Type;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.*;

public final class ModuleRegistry {
    private static final List<ClientModule> MODULES = new ArrayList<>();
    private static final Gson GSON = new Gson();
    private static final Type LIST_TYPE = new TypeToken<List<Map<String, Object>>>() {}.getType();

    private ModuleRegistry() {}

    public static void loadCatalog() {
        try (Reader reader = ModuleRegistry.class.getResourceAsStream("/modules.json") == null
                ? null : new java.io.InputStreamReader(ModuleRegistry.class.getResourceAsStream("/modules.json"))) {
            if (reader == null) throw new IllegalStateException("Missing modules.json");
            List<Map<String, Object>> rows = GSON.fromJson(reader, LIST_TYPE);
            for (Map<String, Object> row : rows) {
                String id = String.valueOf(row.get("id"));
                String name = String.valueOf(row.get("name"));
                String category = String.valueOf(row.get("category"));
                String description = String.valueOf(row.get("description"));
                boolean implemented = "implemented".equals(row.get("state"));
                MODULES.add(new ClientModule(id, name, category, description, implemented));
            }
        } catch (Exception e) {
            throw new IllegalStateException("Could not load module catalog", e);
        }
    }

    public static List<ClientModule> all() {
        return Collections.unmodifiableList(MODULES);
    }

    public static List<ClientModule> inCategory(String category) {
        return MODULES.stream().filter(m -> m.category().equals(category)).toList();
    }

    public static void onStateChanged(ClientModule module) {
        ClientConfig.save();
        ClientFeatures.onModuleChanged(module);
    }

    public static Optional<ClientModule> byId(String id) {
        return MODULES.stream().filter(m -> m.id().equals(id)).findFirst();
    }
}
